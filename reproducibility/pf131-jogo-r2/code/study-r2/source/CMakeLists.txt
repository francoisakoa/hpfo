cmake_minimum_required(VERSION 3.16)
project(pf_cpp VERSION 1.3.1 LANGUAGES CXX)
option(PF_WITH_METIS "Explicit METIS ordering for native LDL and MUMPS" OFF)
option(PF_WITH_MUMPS "Native direct Newton via sequential MUMPS" OFF)
option(PF_WITH_IPOPT "Build the direct Ipopt interface" ON)
option(PF_WITH_OPENMP "Parallel batch evaluation and clustering" ON)
option(PF_NATIVE "Optimize for the build machine (non-portable binary)" OFF)
option(PF_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
include(CTest)
include(GNUInstallDirs)

find_package(Eigen3 3.3 QUIET NO_MODULE)
if(NOT Eigen3_FOUND)
  find_path(PF_EIGEN_INCLUDE Eigen/Core HINTS ${EIGEN3_INCLUDE_DIR} $ENV{CONDA_PREFIX}/include/eigen3 PATH_SUFFIXES eigen3)
  if(NOT PF_EIGEN_INCLUDE)
    message(FATAL_ERROR "Eigen3 headers required. Set -DPF_EIGEN_INCLUDE=/path/to/include/eigen3")
  endif()
  add_library(Eigen3::Eigen INTERFACE IMPORTED)
  set_target_properties(Eigen3::Eigen PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${PF_EIGEN_INCLUDE}")
endif()
add_library(pf STATIC src/ordering.cpp src/linear.cpp src/lj_newton.cpp src/random.cpp src/seed.cpp src/problem.cpp src/operators.cpp src/clustering.cpp src/local.cpp src/ipopt.cpp src/local_pool.cpp src/solver.cpp src/io.cpp)
target_compile_features(pf PUBLIC cxx_std_17)
target_include_directories(pf PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(pf PUBLIC Eigen3::Eigen PRIVATE ${CMAKE_DL_LIBS})
target_compile_definitions(pf PUBLIC EIGEN_DONT_PARALLELIZE)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(pf PUBLIC -fno-fast-math -ffp-contract=off)
  target_compile_options(pf PRIVATE -Wall -Wextra -Wpedantic -Wno-deprecated-declarations)
  if(PF_NATIVE)
    # Public API objects own Eigen memory. Every consumer must use the same
    # ISA/alignment/allocator definitions as the library, including destructors.
    target_compile_options(pf PUBLIC -march=native)
  endif()
  if(PF_SANITIZERS)
    target_compile_options(pf PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(pf PUBLIC -fsanitize=address,undefined)
  endif()
endif()
if(PF_WITH_OPENMP)
  find_package(OpenMP REQUIRED)
  target_link_libraries(pf PUBLIC OpenMP::OpenMP_CXX)
endif()
if(PF_WITH_IPOPT)
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(IPOPT QUIET IMPORTED_TARGET ipopt)
  endif()
  if(TARGET PkgConfig::IPOPT)
    target_link_libraries(pf PRIVATE PkgConfig::IPOPT)
  else()
    find_path(IPOPT_INCLUDE_DIR IpStdCInterface.h HINTS ${IPOPT_ROOT} $ENV{CONDA_PREFIX} PATH_SUFFIXES include/coin-or include/coin coin-or coin)
    find_library(IPOPT_LIBRARY NAMES ipopt HINTS ${IPOPT_ROOT} $ENV{CONDA_PREFIX} PATH_SUFFIXES lib lib64)
    if(NOT IPOPT_INCLUDE_DIR OR NOT IPOPT_LIBRARY)
      message(FATAL_ERROR "Ipopt headers/library required. Set IPOPT_ROOT or IPOPT_INCLUDE_DIR and IPOPT_LIBRARY, or explicitly set PF_WITH_IPOPT=OFF")
    endif()
    target_include_directories(pf PRIVATE ${IPOPT_INCLUDE_DIR})
    target_link_libraries(pf PRIVATE ${IPOPT_LIBRARY})
  endif()
  target_compile_definitions(pf PRIVATE PF_WITH_IPOPT)
endif()
if(PF_WITH_MUMPS)
  find_path(PF_MUMPS_INCLUDE dmumps_c.h HINTS ${PF_MUMPS_ROOT} $ENV{CONDA_PREFIX} PATH_SUFFIXES include include/mumps include/coin-or/mumps)
  find_library(PF_MUMPS_LIBRARY NAMES dmumps_seq coinmumps HINTS ${PF_MUMPS_ROOT} $ENV{CONDA_PREFIX} PATH_SUFFIXES lib lib64)
  if(NOT PF_MUMPS_INCLUDE OR NOT PF_MUMPS_LIBRARY)
    message(FATAL_ERROR "Sequential MUMPS headers/library required; set PF_MUMPS_INCLUDE and PF_MUMPS_LIBRARY")
  endif()
  target_include_directories(pf PRIVATE ${PF_MUMPS_INCLUDE})
  target_link_libraries(pf PRIVATE ${PF_MUMPS_LIBRARY})
  target_compile_definitions(pf PRIVATE PF_WITH_MUMPS)
endif()
if(PF_WITH_METIS)
  find_path(PF_METIS_INCLUDE metis.h HINTS ${PF_METIS_ROOT} $ENV{CONDA_PREFIX} PATH_SUFFIXES include include/metis include/coin-or/metis)
  find_library(PF_METIS_LIBRARY NAMES metis coinmetis HINTS ${PF_METIS_ROOT} $ENV{CONDA_PREFIX} PATH_SUFFIXES lib lib64)
  if(NOT PF_METIS_INCLUDE OR NOT PF_METIS_LIBRARY)
    message(FATAL_ERROR "METIS headers/library required; set PF_METIS_INCLUDE and PF_METIS_LIBRARY (matching ABI)")
  endif()
  target_include_directories(pf SYSTEM PRIVATE ${PF_METIS_INCLUDE})
  target_link_libraries(pf PRIVATE ${PF_METIS_LIBRARY})
  target_compile_definitions(pf PRIVATE PF_WITH_METIS)
endif()
add_executable(pf_solver app/main.cpp)
target_link_libraries(pf_solver PRIVATE pf)
add_executable(pf_local_worker app/worker.cpp)
target_include_directories(pf_local_worker PRIVATE src)
target_link_libraries(pf_local_worker PRIVATE pf)
add_dependencies(pf_solver pf_local_worker)
add_executable(pf_local_bench app/local_bench.cpp)
target_link_libraries(pf_local_bench PRIVATE pf)
add_dependencies(pf_local_bench pf_local_worker)
add_executable(pf_linear_check app/linear_check.cpp)
target_link_libraries(pf_linear_check PRIVATE pf)
if(BUILD_TESTING)
  add_executable(pf_local_incumbent_tests tests/test_local_incumbent.cpp)
  target_link_libraries(pf_local_incumbent_tests PRIVATE pf)
  add_dependencies(pf_local_incumbent_tests pf_local_worker)
  add_test(NAME local_incumbent COMMAND pf_local_incumbent_tests)
  set_tests_properties(local_incumbent PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=1;OPENBLAS_NUM_THREADS=1;MKL_NUM_THREADS=1")
  add_test(NAME linear_ldl COMMAND pf_linear_check --backend ldl)
  if(PF_WITH_METIS)
    add_test(NAME linear_metis_ldl COMMAND pf_linear_check --backend ldl --ordering metis)
  endif()
  if(PF_WITH_MUMPS)
    add_test(NAME linear_mumps COMMAND pf_linear_check --backend mumps)
    if(PF_WITH_METIS)
      add_test(NAME linear_metis_mumps COMMAND pf_linear_check --backend mumps --ordering metis)
    endif()
  endif()
  add_executable(pf_tests tests/test_core.cpp)
  target_link_libraries(pf_tests PRIVATE pf)
  add_dependencies(pf_tests pf_local_worker)
  add_test(NAME core COMMAND pf_tests)
  set_tests_properties(core PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=1;OPENBLAS_NUM_THREADS=1;MKL_NUM_THREADS=1")
  add_executable(pf_eigen_boundary_tests tests/test_eigen_boundary.cpp)
  target_link_libraries(pf_eigen_boundary_tests PRIVATE pf)
  add_test(NAME eigen_memory_boundary COMMAND pf_eigen_boundary_tests)
  add_executable(pf_probe tests/probe.cpp)
  target_link_libraries(pf_probe PRIVATE pf)
  add_executable(pf_lj_newton_tests tests/test_lj_newton.cpp)
  target_link_libraries(pf_lj_newton_tests PRIVATE pf)
  add_test(NAME lj_newton COMMAND pf_lj_newton_tests)
  add_executable(pf_local_probe tests/local_probe.cpp)
  target_link_libraries(pf_local_probe PRIVATE pf)
  add_dependencies(pf_local_probe pf_local_worker)
  add_executable(pf_rng_probe tests/rng_probe.cpp)
  target_link_libraries(pf_rng_probe PRIVATE pf)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME numpy_random_parity COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/numpy_random_parity.py $<TARGET_FILE:pf_rng_probe> $<TARGET_FILE:pf_solver> --report ${CMAKE_CURRENT_BINARY_DIR}/numpy_random_parity.json)
    add_test(NAME python_source_parity COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/python_parity.py $<TARGET_FILE:pf_probe>)
    add_test(NAME cli_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:pf_solver>)
  endif()
endif()
install(TARGETS pf pf_solver pf_local_worker pf_local_bench pf_linear_check ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES README.md THIRD_PARTY_NOTICES.md docs/RANDOM_PROTOCOL.md docs/PORTAGE.md docs/LJ_NEWTON.md docs/METIS.md docs/LOCAL_INCUMBENT.md docs/INSTALL_TEST_COMMANDS.md DESTINATION ${CMAKE_INSTALL_DATADIR}/pf_cpp)

  add_test(NAME article_study_controls COMMAND pf_study_controls_tests)
